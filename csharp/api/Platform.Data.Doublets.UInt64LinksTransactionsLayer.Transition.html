private enum TransitionType { Creation, UpdateOf, UpdateTo, Deletion }

public struct TransitionHeader { public TLinkAddress TransactionIdCombined; public TLinkAddress TimestampCombined;

{
    get
    {
        return (TLinkAddress) mask & TransactionIdCombined;
    }
}

public UniqueTimestamp Timestamp
{
    get
    {
        return (UniqueTimestamp)mask & TransactionIdCombined;
    }
}

}
